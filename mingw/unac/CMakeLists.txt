cmake_minimum_required(VERSION 3.16)
project(reactos_unac C)
set(CMAKE_C_STANDARD 99)
add_library(iconv SHARED IMPORTED)
set_target_properties(iconv PROPERTIES IMPORTED_IMPLIB "${STAGE}/library/libiconv.dll.a")
add_library(unac SHARED "${WORKSPACE}/port/base/unac/unac.c")
set_target_properties(unac PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(unac PRIVATE "${WORKSPACE}/port/base/unac/include" "${WORKSPACE}/port/base/iconv/include")
target_compile_definitions(unac PRIVATE __REACTOS__ _WIN32_WINNT=0x0501 WINVER=0x0501 HAVE_VSNPRINTF UNAC_VERSION="1.8.0")
target_link_options(unac PRIVATE -static-libgcc)
target_link_libraries(unac PRIVATE iconv)
install(TARGETS unac RUNTIME DESTINATION binary ARCHIVE DESTINATION library)
install(FILES "${WORKSPACE}/port/base/unac/include/unac.h" DESTINATION include)
