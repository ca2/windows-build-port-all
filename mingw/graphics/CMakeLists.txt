cmake_minimum_required(VERSION 3.17)
get_filename_component(mingw_bin "${CMAKE_C_COMPILER}" DIRECTORY)
set(CMAKE_RC_COMPILER "${mingw_bin}/windres.exe" CACHE FILEPATH "MinGW resource compiler" FORCE)
project(reactos_graphics C)
set(PLATFORM_NAME windows)
set(LINUX FALSE)
set(__BSD__ FALSE)
set(__ANDROID__ FALSE)
set(BUILD_SHARED_LIBS ON)
set(DISABLE_FORCE_DEBUG_POSTFIX ON)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -D__REACTOS__ -D_WIN32_WINNT=0x0501 -DWINVER=0x0501")
set(os_config "${WORKSPACE}/operating_system/operating_system-windows")
include_directories("${os_config}" "${os_config}/configuration/common" "${STAGE}/include")
add_library(zlib SHARED IMPORTED GLOBAL)
set_target_properties(zlib PROPERTIES IMPORTED_IMPLIB "${STAGE}/library/libzlib.dll.a")
file(GLOB png_sources "${WORKSPACE}/port/imaging/port_png/png*.c")
add_library(png SHARED ${png_sources})
target_include_directories(png PRIVATE "${WORKSPACE}/port/imaging/port_png/include" "${WORKSPACE}/port/imaging/port_png")
target_include_directories(png PRIVATE "${os_config}/configuration/port_png" "${os_config}/configuration" "${WORKSPACE}/port/imaging/port_png/include/port_png")
target_compile_definitions(png PRIVATE PNG_BUILD_DLL)
target_link_libraries(png PRIVATE zlib)
add_subdirectory("${WORKSPACE}/port/graphics/freetype" freetype)
set(PIXMAN_VERSION_MAJOR 0)
set(PIXMAN_VERSION_MINOR 40)
set(PIXMAN_VERSION_MICRO 0)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/pixman")
configure_file("${WORKSPACE}/port/graphics/pixman/pixman/pixman-version.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/pixman/pixman-version.h" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated" "${WORKSPACE}/port/graphics/pixman/include")
add_subdirectory("${WORKSPACE}/port/graphics/pixman" pixman)
file(READ "${os_config}/configuration/pixman/config.h" pixman_config)
string(REGEX REPLACE "#define USE_[A-Z0-9_]+ 1" "" pixman_config "${pixman_config}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pixman/config.h" "${pixman_config}")
target_include_directories(pixman BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/pixman")
set_target_properties(static_pixman PROPERTIES EXCLUDE_FROM_ALL TRUE)

file(COPY "${WORKSPACE}/port/graphics/cairo/include/cairo/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/generated/cairo")
file(READ "${CMAKE_CURRENT_BINARY_DIR}/generated/cairo/cairo-features.h" features)
string(REPLACE "#define CAIRO_HAS_GL_SURFACE 1" "" features "${features}")
string(REPLACE "#define CAIRO_HAS_WGL_FUNCTIONS 1" "" features "${features}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/cairo/cairo-features.h" "${features}")
include_directories(BEFORE "${CMAKE_CURRENT_BINARY_DIR}/generated/cairo")
set(library_include_directories "${CMAKE_CURRENT_BINARY_DIR}/generated/cairo" "${WORKSPACE}/port/graphics/pixman/include/pixman" "${WORKSPACE}/port/graphics/freetype/include" "${WORKSPACE}/port/imaging/port_png/include" "${WORKSPACE}/port/imaging/port_png/include/port_png" "${os_config}/configuration")
set(HAS_PNG ON)
add_subdirectory("${WORKSPACE}/port/graphics/cairo" cairo)
target_link_libraries(cairo PRIVATE pixman freetype png zlib msimg32 gdi32 user32)
foreach(component png pixman freetype cairo)
    set_target_properties(${component} PROPERTIES PREFIX "" DEBUG_POSTFIX "")
    install(TARGETS ${component} RUNTIME DESTINATION binary ARCHIVE DESTINATION library)
endforeach()
install(DIRECTORY "${WORKSPACE}/port/graphics/freetype/include/" DESTINATION include)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/cairo/" DESTINATION include/cairo)
install(DIRECTORY "${WORKSPACE}/port/graphics/pixman/include/pixman/" DESTINATION include/pixman)
install(DIRECTORY "${WORKSPACE}/port/imaging/port_png/include/" DESTINATION include)

install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/pixman/pixman-version.h" DESTINATION include/pixman)
install(FILES "${os_config}/configuration/pngconf.h" "${os_config}/configuration/pnglibconf.h" DESTINATION include)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/freetype/include/freetype/config/ftconfig.h" "${CMAKE_CURRENT_BINARY_DIR}/freetype/include/freetype/config/ftoption.h" DESTINATION include/freetype/config)
