cmake_minimum_required(VERSION 3.16)
project(reactos_pcre C)
set(CMAKE_C_STANDARD 99)
set(pcre_source "${WORKSPACE}/port/base/pcre")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#pragma once\n#include <stddef.h>\n#include <stdint.h>\n#define _(text) (text)\nchar *str_iconv(const char *, const char *, const char *);\n#define HAVE_ICONV 1\n#define ICONV_CONST\n#define HAVE_STDINT_H 1\n#define HAVE_STRING_H 1\n#define HAVE_STDLIB_H 1\n#define HAVE_LOCALE_H 1\n#define c_strcasecmp _stricmp\n#define c_strncasecmp _strnicmp\n#define _GL_ATTRIBUTE_PURE\n#define _GL_ATTRIBUTE_CONST\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/configmake.h" "#define LIBDIR \".\"\n")
target_include_directories(iconv PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${iconv_source}/include")
set(idn_sources idn-free.c idna.c nfkc.c pr29.c profiles.c punycode.c rfc3454.c
 strerror-idna.c strerror-pr29.c strerror-punycode.c strerror-stringprep.c
 strerror-tld.c stringprep.c tld.c tlds.c toutf8.c gl/striconv.c)
list(TRANSFORM idn_sources PREPEND "${idn_source}/lib/")
add_library(idn SHARED ${idn_sources})
target_compile_definitions(idn PRIVATE HAVE_CONFIG_H LIBIDN_BUILDING WINDOWS)
target_include_directories(idn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${WORKSPACE}/operating_system/operating_system-windows" "${idn_source}/include" "${idn_source}/lib" "${idn_source}/lib/gl" "${iconv_source}/include")
target_link_libraries(idn PRIVATE iconv)
foreach(component iconv idn)
 target_link_options(${component} PRIVATE -static-libgcc)
 set_target_properties(${component} PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
 target_compile_definitions(${component} PRIVATE __REACTOS__ _WIN32_WINNT=0x0501 WINVER=0x0501)
endforeach()
install(TARGETS iconv idn RUNTIME DESTINATION binary ARCHIVE DESTINATION library)
install(DIRECTORY "${idn_source}/include/idn" DESTINATION include)
install(FILES "${idn_source}/lib/idn-free.h" "${idn_source}/lib/stringprep.h" "${idn_source}/lib/punycode.h" DESTINATION include)
install(FILES "${iconv_source}/include/iconv.h" DESTINATION include)
